# Fuzz Testing CMake Configuration
# Enable with: cmake -B build -DNOVA_BUILD_FUZZ_TESTS=ON

option(NOVA_BUILD_FUZZ_TESTS "Build fuzz testing targets" OFF)

if(NOT NOVA_BUILD_FUZZ_TESTS)
    return()
endif()

# Check for libFuzzer (provided by the Clang driver's resource directory)
set(FUZZER_LIB "")
if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    set(FUZZER_LIB "Fuzzer")
elseif(CMAKE_CUDA_COMPILER_ID STREQUAL "NVIDIA")
    # CUDA uses clang-based driver on Clang toolchains
    if(CMAKE_CUDA_COMPILER MATCHES "clang")
        set(FUZZER_LIB "Fuzzer")
    endif()
endif()

# Failing here rather than returning quietly is deliberate. -DNOVA_BUILD_FUZZ_TESTS=ON
# is an explicit request; a configure that answers it with a STATUS line and exit 0
# looks identical to a successful run that built nothing, so a CI step or a developer
# checking "did the fuzzers build?" got a green answer either way. Configure on this
# machine with the option on and GCC as the compiler:
#
#     $ cmake -DNOVA_BUILD_FUZZ_TESTS=ON ...
#     -- Fuzz testing requires Clang/LLVM. Skipping fuzz targets.
#     $ echo $?
#     0
if(NOT FUZZER_LIB)
    message(FATAL_ERROR
        "NOVA_BUILD_FUZZ_TESTS=ON requires a Clang toolchain, because libFuzzer is "
        "compile-time instrumentation. This configure is using "
        "${CMAKE_CXX_COMPILER_ID} (${CMAKE_CXX_COMPILER}). Configure with "
        "-DCMAKE_CXX_COMPILER=clang++ (and a matching CUDA host compiler) to build the "
        "fuzz targets, or configure without the option.")
endif()

# No find_library here. This used to search for a library literally named `Fuzzer`
# under /usr/lib and /usr/local/lib, which is not where any distribution puts it:
# LLVM ships the runtime as a Clang resource, libclang_rt.fuzzer-<arch>.a, invoked by
# the driver through -fsanitize=fuzzer. On a host that has a working libFuzzer, the
# search failed anyway and the configure said the runtime was missing:
#
#     $ cmake -DNOVA_BUILD_FUZZ_TESTS=ON -DCMAKE_CXX_COMPILER=clang++-18 ...
#     CMake Error: needs the libFuzzer runtime (Fuzzer), which was not found
#     $ ls /usr/lib/llvm-18/lib/clang/18/lib/linux/libclang_rt.fuzzer-x86_64.a
#     ...exists
#
# The check was also not load-bearing: the targets below link with
# -fsanitize=fuzzer, and the driver resolves its own runtime. Let the compiler decide
# -- a Clang toolchain either has the resource or the link fails with a real message.

message(STATUS "Building fuzz testing targets with libFuzzer")

# Memory Pool Fuzz Target
add_executable(memory_pool_fuzz
    fuzz/memory_pool_fuzz.cpp
)
target_link_libraries(memory_pool_fuzz PRIVATE
    cuda_impl
    CUDA::cudart
)
target_include_directories(memory_pool_fuzz PRIVATE
    ${CMAKE_SOURCE_DIR}/include
    ${CMAKE_SOURCE_DIR}/tests/fuzz
)
target_compile_options(memory_pool_fuzz PRIVATE -fsanitize=fuzzer)
# The link side needs the flag too. -fsanitize=fuzzer is compile-time instrumentation
# that emits references to the libFuzzer runtime (__sanitizer_cov_*), and the driver
# only adds that runtime when the flag is present while LINKING. It was set on compile
# only, so linking failed with:
#
#   mold: error: undefined symbol: __sanitizer_cov_pcs_init
#   mold: error: undefined symbol: __sanitizer_cov_trace_const_cmp4
#
# ${FUZZER_LIBRARY} used to sit in the link list for this purpose, but it was only
# ever set by a find_library that searched for a file named libFuzzer.a, which no
# distribution ships -- LLVM's runtime is a Clang resource, libclang_rt.fuzzer-*.a.
# It was therefore always empty, which is why the gap survived.
target_link_options(memory_pool_fuzz PRIVATE -fsanitize=fuzzer)

add_custom_target(fuzz_memory_pool
    COMMAND memory_pool_fuzz $<TARGET_FILE:memory_pool_fuzz> ${CMAKE_SOURCE_DIR}/tests/fuzz/corpus/memory_pool -max_total_time=60
    DEPENDS memory_pool_fuzz
    COMMENT "Running memory pool fuzzing tests"
    VERBATIM
)
